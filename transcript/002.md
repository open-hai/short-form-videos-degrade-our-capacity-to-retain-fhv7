452	We analyzed the reaction time distribution for the correct and

454	the behavioral differences between correct and error responses,
455	we adopted a Drift-Diffusion Model (DDM) [ 86], which has been
456	leveraged in cognitive science [ 46] and computer graphics [ 30] to
457	model perceptual decision-making. The DDM model also has been
458	suggested to model choice andnon-choice in LD and PM tasks [ 11,
459	20,115]. We chose this modeling approach as it is informative
460	on different processing components relevant to PM. Traditional
461	analysis approaches investigated RTs and task accuracy separately,
462	profiting of only a subset of the available data, while the diffusion
463	model is applied to the joint distribution of RT and accuracy data.
464	DDM allows for the decomposition of RTs and accuracy into a set
465	of latent parameters that represent underlying cognitive processes
466	such as task load [ 7], evidence accumulation [ 106], and decreased
467	cognitive capacity [ 110]. The DDM assumes that decisions are made
468	by a Wiener process that accumulates cognitive evidence over time
469	from a starting point towards two response boundaries.A DDM model includes 4 parameters: the drift rate ( 𝜇), the de-
470	cision bound ( 𝐵), the non-decision time ( 𝑡) and variance ( 𝜎) com-
471	ponent. As an interpretation, the drift rate informs the speed and
472	direction of information accumulation. The drift rate can be inter-
473	preted as a measure of subjective task difficulty: higher (absolute)

def fit_ex_gaussian(x: np.ndarray, epochs=500, progress=False) -> Tuple[float, float, float]:
    """
    Fit an expotential Gaussian distribution to the data using Tensorflow Probability.

    An exponential Gaussian distribution is a combination of an exponential
    distribution and a Gaussian distribution which can be modeled using numpy:
    np.random.exponential(scale=1/lambda, size=n) + np.random.normal(loc=location, scale=scale, size=n)

    This function returns the parameters of the exponential Gaussian distribution: location, scale, and lambda.
    """
    tqdm_callback = tfa.callbacks.TQDMProgressBar(show_epoch_progress=False)
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(3),
        tfp.layers.DistributionLambda(
            lambda t: tfp.distributions.ExponentiallyModifiedGaussian(
                loc=t[..., :1],
                scale=1e-3 + tf.math.softplus(0.05 * t[..., 1:]),
                rate=tf.math.softplus(t[..., 2:]),
            )
        ),
    ])
    lr_schedule = tf.keras.optimizers.schedules.ExponentialDecay(
        0.2, 10, 0.99, staircase=False, name=None
    )
    model.compile(optimizer=tf.optimizers.Adam(learning_rate=lr_schedule),
                  loss=lambda y, distr: -distr.log_prob(y))
    model.fit(np.zeros(x.shape)[:, np.newaxis], x,
              epochs=epochs, verbose=0, callbacks=None if not progress else [tqdm_callback])
    return model.weights[1][0], tf.math.softplus(0.05 * model.weights[1][1]), tf.math.softplus(model.weights[1][2])
